CREATE TABLE student (
    student_id INT PRIMARY KEY AUTO_INCREMENT,

    -- PERSONAL DETAILS
    first_name      VARCHAR(50) NOT NULL,
    last_name       VARCHAR(50) NOT NULL,
    email           VARCHAR(100) UNIQUE NOT NULL,
    contact_no      VARCHAR(14),
    gender          ENUM('M', 'F', 'O') NOT NULL DEFAULT 'O',
    class           ENUM('1st','2nd','3rd','4th','5th','6th','7th','8th','9th','10th') NOT NULL,
    age             INT CHECK (age >= 5 AND age <= 18),
    weight          FLOAT CHECK (weight > 0),
    blood_group     VARCHAR(7) CHECK (blood_group IN ('A+', 'A-', 'B+', 'B-', 'AB+', 'AB-', 'O+', 'O-')),
    religion        VARCHAR(40),
    address         VARCHAR(255),
    city            VARCHAR(50),
    pincode         INT CHECK (pincode BETWEEN 100000 AND 999999),
    passing_status  BOOLEAN DEFAULT TRUE,
    admission_date  DATE NOT NULL,
    
    -- ACADEMIC DETAILS
    highest_education       VARCHAR(100),
    current_course          VARCHAR(100),
    current_year_semester   VARCHAR(50),
    school_college_name     VARCHAR(150),
    percentage_or_cgpa      VARCHAR(10),

    -- PROFESSIONAL DETAILS
    has_experience      BOOLEAN DEFAULT FALSE,
    experience_years    FLOAT DEFAULT 0
                        CHECK (experience_years >= 0 AND experience_years <= 50),
    current_job_role    VARCHAR(100),
    skills              VARCHAR(255),
    fees                DECIMAL(10,2) NOT NULL CHECK (fees > 0),
    created_at          TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

🔗 One-to-One Relationship (1 : 1) — Simple Explanation
🧠 Concept

One-to-One relationship ka matlab:

👉 Table A ka ek record
👉 Table B ke sirf ek record se linked hoga
Na zyada, na kam.


🎯 Real Life Example: Person and Passport
✔ Ek person ka sirf ek passport hota hai
✔ Ek passport sirf ek person ka hota hai

Person 1  ⟷  1 Passport


✅ STEP 1: Structure (Tables Design)
We create 2 tables:

person
passport



👨 Person Table
CREATE TABLE person (
    person_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    age INT
);



🛂 Passport Table
⚠ Important: person_id must be UNIQUE
CREATE TABLE passport (
    passport_id INT AUTO_INCREMENT PRIMARY KEY,
    passport_number VARCHAR(20) UNIQUE,
    person_id INT UNIQUE,

    FOREIGN KEY (person_id)
    REFERENCES person(person_id)
    ON DELETE CASCADE
);



🧠 Why UNIQUE is used?
Because:
One person → one passport only

UNIQUE ensures:
❌ One person cannot have 2 passports


✅ Final Structure
person
-------
person_id (PK)
name
age


passport
--------
passport_id (PK)
passport_number
person_id (FK + UNIQUE)

3️⃣ Many-to-Many Relationship
🎯 Example:

Student can join multiple courses
Course can have multiple students

✅ Structure
We need 3 tables

Student Table
(already created)

Course Table
CREATE TABLE courses (
    course_id INT AUTO_INCREMENT PRIMARY KEY,
    course_name VARCHAR(100)
);



Junction Table
CREATE TABLE student_course (
    id INT AUTO_INCREMENT PRIMARY KEY,
    student_id INT,
    course_id INT,

    FOREIGN KEY (student_id)
    REFERENCES student(student_id),

    FOREIGN KEY (course_id)
    REFERENCES courses(course_id)
);



✅ Insert
INSERT INTO courses (course_name) VALUES ('Python'), ('SQL');
INSERT INTO student_course (student_id, course_id) VALUES (1,1), (1,2), (2,1);



✅ Query
SELECT student.name, courses.course_name
FROM student_course

JOIN student
ON student.student_id = student_course.student_id

JOIN courses
ON courses.course_id = student_course.course_id;
